# R3 LLM Call Cache Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: superpowers:subagent-driven-development.

**Goal:** Add `tmp/.cache/llm/<sha256>.json` content-addressed cache for LLM JSON completions. Hash key = (model, system_prompt, user_payload_canonical_json). Deterministic given inputs → permanent cache hit until catalog/model change naturally bumps the hash. No TTL.

**Architecture:** New `cache/llm_cache.py` with `cache_key()` + `cache_get()` + `cache_put()`. Three transport clients in `llm_transport.py` (`OpenAiCompatibleClient`, `CodexResponsesClient`, `ClaudeCodeMessagesClient`) consult the cache before HTTP call. CLI flag `--no-llm-cache` bypasses. Cache file stores `{request: {...}, response: {...}}` so audit trails are preserved.

**Tech Stack:** Python 3.11 stdlib (`hashlib`, `json`), pytest. Zero new deps.

---

## Key Design Decisions

1. **Hash key**: SHA-256 of `json.dumps({"model": ..., "system_prompt": ..., "user_payload": ...}, sort_keys=True, ensure_ascii=False)`. Stable across runs given same inputs.
2. **No TTL**: deterministic — given inputs, the response shouldn't change. Catalog change → chunks differ → hash differs → automatic miss.
3. **Backward compat**: existing tests using mock transports must not break. Cache is opt-in via `cache_root` parameter on the client constructor; default `None` = no caching.
4. **Three providers**: OpenAI-compatible, Codex Responses (SSE), Claude Code. All share the same cache layer (key includes provider via the `model` field — different providers use distinct model identifiers).
5. **Cache stores the full raw response** (provider-native shape). The existing `_parse_response_json_content()` helpers continue to interpret it.

### Step 1.3: Implement `llm_cache.py`

```python
"""Content-addressed cache for LLM JSON completions (R3 layer-2).

Cache key: SHA-256 of canonical-JSON of (model, system_prompt, user_payload).
No TTL — given identical inputs the response should not change. Catalog
changes propagate via chunk content → hash → cache miss → re-fetch.

Cache file layout: tmp/.cache/llm/<sha256>.json
File format:
  {
    "schema_version": "llm_cache_v1",
    "key": "<sha256>",
    "cached_at": "<ISO8601 UTC>",
    "request": {...},
    "response": {...}
  }

Malformed cache files → silent miss (next cache_put overwrites).
"""

from __future__ import annotations

import hashlib
import json
from datetime import datetime, timezone
from pathlib import Path
from typing import Any

SCHEMA_VERSION = "llm_cache_v1"


def cache_key(
    *,
    model: str,
    system_prompt: str,
    user_payload: dict[str, Any],
) -> str:
    """SHA-256 hex digest of the canonical-JSON request triple."""
    canonical = json.dumps(
        {
            "model": model,
            "system_prompt": system_prompt,
            "user_payload": user_payload,
        },
        sort_keys=True,
        ensure_ascii=False,
        allow_nan=False,
    )
    return hashlib.sha256(canonical.encode("utf-8")).hexdigest()


def cache_path(*, cache_root: Path, key: str) -> Path:
    """Return path tmp_cache_root/llm/<key>.json."""
    if len(key) != 64 or not all(c in "0123456789abcdef" for c in key):
        raise ValueError(f"key must be a 64-char hex SHA-256 digest, got {key!r}")
    return cache_root / "llm" / f"{key}.json"


def cache_get(
    *,
    cache_root: Path,
    key: str,
) -> dict[str, Any] | None:
    """Return cached entry dict ({request, response}) or None on miss/malformed."""
    path = cache_path(cache_root=cache_root, key=key)
    if not path.exists():
        return None
    try:
        payload = json.loads(path.read_text(encoding="utf-8"))
    except (json.JSONDecodeError, OSError):
        return None
    if not isinstance(payload, dict):
        return None
    if payload.get("schema_version") != SCHEMA_VERSION:
        return None
    request = payload.get("request")
    response = payload.get("response")
    if not isinstance(request, dict) or not isinstance(response, dict):
        return None
    return {"request": request, "response": response}


def cache_put(
    *,
    cache_root: Path,
    key: str,
    request: dict[str, Any],
    response: dict[str, Any],
) -> None:
    """Write request+response to the cache."""
    path = cache_path(cache_root=cache_root, key=key)
    path.parent.mkdir(parents=True, exist_ok=True)
    payload = {
        "schema_version": SCHEMA_VERSION,
        "key": key,
        "cached_at": datetime.now(timezone.utc).isoformat(),
        "request": request,
        "response": response,
    }
    path.write_text(
        json.dumps(payload, ensure_ascii=False, indent=2, sort_keys=True),
        encoding="utf-8",
    )
```

```python
def complete_json(
    self,
    *,
    system_prompt: str,
    user_payload: dict[str, object],
) -> dict[str, object]:
    # Cache lookup
    if self.cache_root is not None:
        from financial_report_llm_extractor.cache.llm_cache import (
            cache_get as _llm_cache_get,
            cache_key as _llm_cache_key,
        )
        key = _llm_cache_key(
            model=self.config.model,
            system_prompt=system_prompt,
            user_payload=user_payload,
        )
        hit = _llm_cache_get(cache_root=self.cache_root, key=key)
        if hit is not None:
            # Record raw exchange for archival (same as live path)
            self.raw_exchanges.append(RawExchange(
                request=hit["request"], raw_response=hit["response"],
            ))
            return hit["response"]

    # Existing live call ...
    payload = {...}
    raw_response = self._post_with_retries(payload)
    self.raw_exchanges.append(RawExchange(request=payload, raw_response=raw_response))

    # Cache write
    if self.cache_root is not None:
        from financial_report_llm_extractor.cache.llm_cache import (
            cache_put as _llm_cache_put,
        )
        _llm_cache_put(
            cache_root=self.cache_root, key=key,
            request=payload, response=raw_response,
        )
    return raw_response
```

Note: For `CodexResponsesClient` and `ClaudeCodeMessagesClient`, the `payload` shape differs from OpenAI — make sure the cache stores the **provider-native** request (Codex Responses payload or Anthropic Messages payload), not a normalized OpenAI shape. The provider-native shape is what the existing code builds inside each `complete_json`.

### Step 2.4: Write integration tests

`tests/test_llm_transport_cache.py`:

```python
"""LLM transport clients integrate with the llm_cache module."""

from pathlib import Path
from unittest.mock import MagicMock

from financial_report_llm_extractor.cache.llm_cache import cache_put, cache_key
from financial_report_llm_extractor.llm_transport import (
    HttpTransport,
    LlmTransportConfig,
    OpenAiCompatibleClient,
)


class _CountingTransport:
    """Tracks how many times post_json was called."""
    def __init__(self, response: dict) -> None:
        self.response = response
        self.calls = 0
    def post_json(self, url, headers, payload, timeout_seconds):
        self.calls += 1
        return self.response


def test_openai_client_skips_http_on_cache_hit(tmp_path: Path) -> None:
    """When cache_root is set and key matches, no HTTP call."""
    config = LlmTransportConfig(
        provider="openai-compatible", model="m",
        base_url="https://example.com", api_key="x",
        timeout_seconds=10, max_retries=0,
    )
    response = {"choices": [{"message": {"content": '{"ok":true}'}}]}

    # Prime cache
    sys_p = "Return JSON."
    user_p = {"field_id": "revenue"}
    key = cache_key(model="m", system_prompt=sys_p, user_payload=user_p)
    # The cache stores the request shape, but the cached request must match
    # what the client constructs. For OpenAI-compatible, the request is the
    # full chat completions payload. Easiest: prime with the actual request
    # the client would build, OR write a placeholder that mimics it.
    cache_put(
        cache_root=tmp_path, key=key,
        request={"model": "m", "messages": "..."},  # placeholder
        response=response,
    )

    transport = _CountingTransport(response={"never": "called"})
    client = OpenAiCompatibleClient(
        config=config, transport=transport, cache_root=tmp_path,
    )
    result = client.complete_json(system_prompt=sys_p, user_payload=user_p)
    assert result == response
    assert transport.calls == 0  # HTTP not called


def test_openai_client_writes_cache_on_miss(tmp_path: Path) -> None:
    """First call hits HTTP, writes cache. Second call hits cache."""
    config = LlmTransportConfig(
        provider="openai-compatible", model="m",
        base_url="https://example.com", api_key="x",
        timeout_seconds=10, max_retries=0,
    )
    live_response = {"choices": [{"message": {"content": '{"x": 1}'}}]}
    transport = _CountingTransport(response=live_response)
    client = OpenAiCompatibleClient(
        config=config, transport=transport, cache_root=tmp_path,
    )
    sys_p = "Return JSON."
    user_p = {"y": 2}
    # First call: cache miss, HTTP called
    r1 = client.complete_json(system_prompt=sys_p, user_payload=user_p)
    assert transport.calls == 1
    assert r1 == live_response
    # Second call (new client instance, same cache_root): cache hit
    client2 = OpenAiCompatibleClient(
        config=config, transport=transport, cache_root=tmp_path,
    )
    r2 = client2.complete_json(system_prompt=sys_p, user_payload=user_p)
    assert r2 == live_response
    assert transport.calls == 1  # unchanged


def test_openai_client_no_cache_when_cache_root_is_none(tmp_path: Path) -> None:
    """Backward compat: cache_root=None means no caching, no behavior change."""
    config = LlmTransportConfig(
        provider="openai-compatible", model="m",
        base_url="https://example.com", api_key="x",
        timeout_seconds=10, max_retries=0,
    )
    transport = _CountingTransport(response={"x": 1})
    client = OpenAiCompatibleClient(
        config=config, transport=transport, cache_root=None,
    )
    client.complete_json(system_prompt="p", user_payload={"q": 1})
    client.complete_json(system_prompt="p", user_payload={"q": 1})
    assert transport.calls == 2  # both calls hit HTTP
```
